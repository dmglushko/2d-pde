endif()

target_include_directories(pde2d_execution INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/cpp")

  find_package(OpenMP REQUIRED COMPONENTS CXX)
  target_link_libraries(pde2d_execution INTERFACE OpenMP::OpenMP_CXX)
endif()
set(PDE2D_SANITIZERS "" CACHE STRING "Optional compiler sanitizer list, e.g. address,undefined or thread")
if(PDE2D_SANITIZERS)
  target_compile_options(pde2d_execution INTERFACE -fsanitize=${PDE2D_SANITIZERS} -fno-omit-frame-pointer)
  target_link_options(pde2d_execution INTERFACE -fsanitize=${PDE2D_SANITIZERS})
endif()
option(PDE2D_BUILD_FEM "Build numerical module" OFF)
option(PDE2D_FETCH_SYMENGINE "Download pinned SymEngine when unavailable" OFF)
option(PDE2D_ENABLE_REFERENCES "Enable isolated Python reference and plot tests" OFF)
set(PDE2D_REFERENCE_PYTHON "${CMAKE_CURRENT_SOURCE_DIR}/.venv/bin/python" CACHE FILEPATH "Isolated reference interpreter")
find_package(Threads REQUIRED)
if(PDE2D_BUILD_FEM)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
# SymEngine 0.14's installed config still declares CMake 2.8 compatibility.
# CMake 4 requires an explicit policy floor when loading such package configs.
if(CMAKE_VERSION VERSION_GREATER_EQUAL 4.0 AND NOT DEFINED CMAKE_POLICY_VERSION_MINIMUM)
  set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
endif()
find_package(SymEngine CONFIG QUIET)

if(NOT SymEngine_FOUND AND NOT SYMENGINE_FOUND)
  if(NOT PDE2D_FETCH_SYMENGINE)
    message(FATAL_ERROR "SymEngine is required; set SymEngine_DIR or enable PDE2D_FETCH_SYMENGINE")
  endif()
  include(FetchContent)
  set(INTEGER_CLASS boostmp CACHE STRING "SymEngine integer backend")
  set(BUILD_TESTS OFF CACHE BOOL "Build SymEngine tests")
  set(BUILD_BENCHMARKS OFF CACHE BOOL "Build SymEngine benchmarks")
  set(WITH_SYMENGINE_RCP ON CACHE BOOL "Use SymEngine reference counting")
  FetchContent_Declare(symengine
    URL https://codeload.github.com/symengine/symengine/tar.gz/refs/tags/v0.14.0
    URL_HASH SHA256=11c5f64e9eec998152437f288b8429ec001168277d55f3f5f1df78e3cf129707)
  FetchContent_MakeAvailable(symengine)
  # Upstream defaults to C++11; recent Boost.Math headers require C++14+.
  set_target_properties(symengine PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
  set(SYMENGINE_INCLUDE_DIRS "${symengine_SOURCE_DIR}" "${symengine_BINARY_DIR}")
  set(SYMENGINE_LIBRARIES symengine)
endif()

add_library(pde2d_fem INTERFACE)

target_link_libraries(pde2d_fem INTERFACE pde2d_execution Eigen3::Eigen ${SYMENGINE_LIBRARIES})

endif()

# Every test gets a private working/fixture directory, including tests added by
# other modules. Export helpers live outside tests/ and cannot become tests.
function(pde2d_test name)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${name}")
  add_test(NAME ${name} COMMAND ${ARGN})
  set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${name}")
endfunction()
if(BUILD_TESTING)
  find_package(LibXml2 QUIET)
  find_program(PDE2D_NODE node)
  file(GLOB test_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp")
  foreach(source IN LISTS test_sources)
    get_filename_component(name "${source}" NAME_WE)
    if(name MATCHES "^fem_" AND NOT PDE2D_BUILD_FEM)
      continue()
    endif()
    if(name MATCHES "svg" AND NOT LibXml2_FOUND)
      message(STATUS "UNAVAILABLE: ${name} (LibXml2)")
      continue()
    endif()
    add_executable(${name} "${source}")
    target_link_libraries(${name} PRIVATE pde2d_execution Threads::Threads)
    if(name MATCHES "^fem_")
      target_link_libraries(${name} PRIVATE pde2d_fem)
    endif()
    if(name MATCHES "svg")
      target_link_libraries(${name} PRIVATE LibXml2::LibXml2)
    endif()
    # Legacy discretization test reads this repository fixture relatively.
    if(name STREQUAL "discretization")
      file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${name}/examples/mesh_viewer")
      configure_file(examples/mesh_viewer/rectangle_p1.pde2d
        "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${name}/examples/mesh_viewer/rectangle_p1.pde2d" COPYONLY)
    endif()
    pde2d_test(${name} $<TARGET_FILE:${name}> "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${name}")
  endforeach()
  foreach(example IN ITEMS learning/first_mesh mesh_viewer/produce_mesh meshing/showcase)
    get_filename_component(name "${example}" NAME)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${name}")
    add_executable(${name} examples/${example}.cpp)
    target_link_libraries(${name} PRIVATE pde2d_execution Threads::Threads)
    if(name STREQUAL "showcase")
      set(output "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${name}/meshes")
    else()
      set(output "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${name}/${name}.pde2d")
    endif()
    pde2d_test(example_${name} $<TARGET_FILE:${name}> "${output}")
    set_tests_properties(example_${name} PROPERTIES FIXTURES_SETUP ${name}_export)
  endforeach()
  if(PDE2D_NODE)
    file(GLOB_RECURSE viewer_tests CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tools/*/test*.cjs")
    foreach(source IN LISTS viewer_tests)
      get_filename_component(parent "${source}" DIRECTORY)
      get_filename_component(parent "${parent}" NAME)
      get_filename_component(name "${source}" NAME_WE)
      pde2d_test(${parent}_${name} "${PDE2D_NODE}" "${source}")
    endforeach()
    pde2d_test(showcase_roundtrip "${PDE2D_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/test_showcase.cjs" "${CMAKE_CURRENT_BINARY_DIR}/fixtures/showcase/meshes")
    set_tests_properties(showcase_roundtrip PROPERTIES FIXTURES_REQUIRED showcase_export)
    pde2d_test(viewer_diagnostics "${CMAKE_COMMAND}" -E chdir "${CMAKE_CURRENT_SOURCE_DIR}" "${PDE2D_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/viewer_diagnostics.cjs")
  else()
    message(STATUS "UNAVAILABLE: JavaScript viewer/roundtrip tests (node)")
  endif()
  if(PDE2D_ENABLE_REFERENCES)
    execute_process(COMMAND "${PDE2D_REFERENCE_PYTHON}" -c "import skfem, scipy, matplotlib" RESULT_VARIABLE reference_status)
    if(NOT reference_status EQUAL 0)
      message(FATAL_ERROR "References requested but isolated dependencies unavailable; install tools/requirements-reference.txt")
    endif()
    pde2d_test(plot_mesh "${PDE2D_REFERENCE_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/test_plot_mesh.py" "${CMAKE_CURRENT_BINARY_DIR}/fixtures/showcase/meshes")
    set_tests_properties(plot_mesh PROPERTIES FIXTURES_REQUIRED showcase_export)
    if(PDE2D_BUILD_FEM)
      add_executable(reference_export tools/reference/export.cpp)
      target_link_libraries(reference_export PRIVATE pde2d_fem)
      pde2d_test(independent_reference "${PDE2D_REFERENCE_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/reference/verify.py" $<TARGET_FILE:reference_export> "${CMAKE_CURRENT_BINARY_DIR}/fixtures/independent_reference")
    else()
      message(STATUS "UNAVAILABLE: independent FEM references (PDE2D_BUILD_FEM=OFF)")
    endif()
  else()
    message(STATUS "UNAVAILABLE: Python independent reference/plot tests (opt in with PDE2D_ENABLE_REFERENCES)")
  endif()
endif()
if(PDE2D_BUILD_FEM)
  foreach(example IN ITEMS showcase long_waves)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/examples/fem/${example}.cpp")
      add_executable(fem_${example} examples/fem/${example}.cpp)
      target_link_libraries(fem_${example} PRIVATE pde2d_fem)
    endif()
  endforeach()
endif()
