cmake_minimum_required(VERSION 3.20)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
project(pde2d LANGUAGES CXX)

include(CTest)
option(PDE2D_ENABLE_OPENMP "Enable explicitly selected OpenMP CPU kernels" OFF)
add_library(pde2d_execution INTERFACE)
target_compile_features(pde2d_execution INTERFACE cxx_std_23)
target_include_directories(pde2d_execution INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/cpp")
if(PDE2D_ENABLE_OPENMP)
  find_package(OpenMP REQUIRED COMPONENTS CXX)
  target_link_libraries(pde2d_execution INTERFACE OpenMP::OpenMP_CXX)
endif()
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
# SymEngine 0.14's installed config still declares CMake 2.8 compatibility.
# CMake 4 requires an explicit policy floor when loading such package configs.
if(CMAKE_VERSION VERSION_GREATER_EQUAL 4.0 AND NOT DEFINED CMAKE_POLICY_VERSION_MINIMUM)
  set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
endif()
find_package(SymEngine CONFIG QUIET)
option(PDE2D_FETCH_SYMENGINE "Build pinned SymEngine if no installation is found" ON)
if(NOT SymEngine_FOUND AND NOT SYMENGINE_FOUND)
  if(NOT PDE2D_FETCH_SYMENGINE)
    message(FATAL_ERROR "SymEngine is required; set SymEngine_DIR or enable PDE2D_FETCH_SYMENGINE")
  endif()
  include(FetchContent)
  set(INTEGER_CLASS boostmp CACHE STRING "SymEngine integer backend")
  set(BUILD_TESTS OFF CACHE BOOL "Build SymEngine tests")
  set(BUILD_BENCHMARKS OFF CACHE BOOL "Build SymEngine benchmarks")
  set(WITH_SYMENGINE_RCP ON CACHE BOOL "Use SymEngine reference counting")
  FetchContent_Declare(symengine
    URL https://codeload.github.com/symengine/symengine/tar.gz/refs/tags/v0.14.0
    URL_HASH SHA256=11c5f64e9eec998152437f288b8429ec001168277d55f3f5f1df78e3cf129707)
  FetchContent_MakeAvailable(symengine)
  # Upstream defaults to C++11; recent Boost.Math headers require C++14+.
  set_target_properties(symengine PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
  set(SYMENGINE_INCLUDE_DIRS "${symengine_SOURCE_DIR}" "${symengine_BINARY_DIR}")
  set(SYMENGINE_LIBRARIES symengine)
endif()

add_library(pde2d_fem INTERFACE)
target_compile_features(pde2d_fem INTERFACE cxx_std_23)
target_include_directories(pde2d_fem INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/cpp")
target_include_directories(pde2d_fem SYSTEM INTERFACE ${SYMENGINE_INCLUDE_DIRS})
target_link_libraries(pde2d_fem INTERFACE pde2d_execution Eigen3::Eigen ${SYMENGINE_LIBRARIES})

if(BUILD_TESTING)
  file(GLOB fem_test_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/fem_*.cpp")
  foreach(source IN LISTS fem_test_sources)
    get_filename_component(name "${source}" NAME_WE)
    add_executable(${name} "${source}")
    target_link_libraries(${name} PRIVATE pde2d_fem)
    add_test(NAME ${name} COMMAND ${name})
  endforeach()
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/examples/fem/showcase.cpp")
  add_executable(fem_showcase examples/fem/showcase.cpp)
  target_link_libraries(fem_showcase PRIVATE pde2d_fem)
endif()
add_executable(fem_long_waves examples/fem/long_waves.cpp)
target_link_libraries(fem_long_waves PRIVATE pde2d_fem)
